#include <SPI.h>
#include <MFRC522.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define RST_PIN 9
#define SS_PIN 10

MFRC522 mfrc522(SS_PIN, RST_PIN);
LiquidCrystal_I2C lcd(0x27, 16, 2);

String tarjetasPermitidas[] = {
  "027DB8DEB6E000",
  "E789B56B",
  "9F3D8CE6",
  "A779C86D", 
  "08DEC7FC",
  "DED0CC05",
  "76F1B105", 
  "C7F9AF05", 
  "CBECB105", 
  "015FB405", 
  "6B61CD05", 
  "ED19CD05", 
  "47F7B005"
};

bool esValida(String uid) {
  for (int i = 0; i < sizeof(tarjetasPermitidas) / sizeof(tarjetasPermitidas[0]); i++) {
    if (tarjetasPermitidas[i] == uid) return true;
  }
  return false;
}

String leerUID() {
  String uidStr = "";
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    if (mfrc522.uid.uidByte[i] < 0x10) uidStr += "0";
    uidStr += String(mfrc522.uid.uidByte[i], HEX);
  }
  uidStr.toUpperCase();  // Convierte correctamente en el mismo String
  return uidStr;
}

void setup() {
  Serial.begin(9600);
  SPI.begin();
  mfrc522.PCD_Init();

  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Listo para leer");
}

void loop() {
  if (!mfrc522.PICC_IsNewCardPresent() || !mfrc522.PICC_ReadCardSerial()) return;

  String uid = leerUID();
  Serial.print("UID leído: ");
  Serial.println(uid);  // Esto te dirá qué UID estás leyendo

  lcd.clear();
  if (esValida(uid)) {
    lcd.print("Acceso permitido");
  } else {
    lcd.print("Acceso denegado");
  }

  mfrc522.PICC_HaltA();
  mfrc522.PCD_StopCrypto1();

  delay(1500);
  lcd.clear();
  lcd.print("Listo para leer");
}
